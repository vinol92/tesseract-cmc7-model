package com.vinol.micr;

import java.io.File;
import net.sourceforge.tess4j.Tesseract;
import net.sourceforge.tess4j.TesseractException;

public class MicrReader {

    public static void main(String[] args) {
        // Ruta de la imagen a leer
        String imagePath = "C:\\Users\\vinol\\Pictures\\prueba.jpg";

        // Crear instancia de Tesseract
        Tesseract tesseract = new Tesseract();

        // Ruta a la carpeta tessdata donde está tu modelo cmc7.traineddata
        tesseract.setDatapath("C:\\Program Files\\Tesseract-OCR\\tessdata");

        // Usar tu modelo personalizado
        tesseract.setLanguage("cmc7");

        // Modo de segmentación de página 7 = una sola línea de texto
        tesseract.setPageSegMode(7);

        try {
            System.out.println("Leyendo número MICR...");
            String result = tesseract.doOCR(new File(imagePath));

            // Limpiar el resultado (opcional)
            result = result.replaceAll("[^0-9A-Za-z ]", "").trim();

            System.out.println("Resultado OCR:");
            System.out.println(result);

        } catch (TesseractException e) {
            e.printStackTrace();
        }
    }
}
